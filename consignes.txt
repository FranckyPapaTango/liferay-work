# version: '3.8'

services:

  db:
    image: postgres:15
    container_name: liferay-db
    environment:
      POSTGRES_DB: liferay
      POSTGRES_USER: liferay
      POSTGRES_PASSWORD: liferay
      POSTGRES_INITDB_ARGS: "--encoding=UTF8"
    volumes:
      - ./postgres:/var/lib/postgresql/data
    healthcheck:
      test: [ "CMD-SHELL", "pg_isready -U liferay -d liferay" ]
      interval: 5s
      timeout: 5s
      retries: 10

  liferay:
  #  image: liferay/dxp:2026.q1.0-lts
    image: liferay/portal:7.4.3.132-ga132
    container_name: liferay
    depends_on:
      db:
        condition: service_healthy
    environment:
      LIFERAY_JDBC_PERIOD_DEFAULT_PERIOD_URL: jdbc:postgresql://db:5432/liferay
      LIFERAY_JDBC_PERIOD_DEFAULT_PERIOD_USERNAME: liferay
      LIFERAY_JDBC_PERIOD_DEFAULT_PERIOD_PASSWORD: liferay
    volumes:
      - ./liferay/deploy:/opt/liferay/deploy
      - ./liferay/data:/opt/liferay/data
      - ./liferay/logs:/opt/liferay/logs
      - ./liferay/portal-ext.properties:/opt/liferay/portal-ext.properties
    ports:
      - "8080:8080"

  nginx:
    image: nginx:latest
    container_name: liferay-nginx
    depends_on:
      - liferay
    ports:
      - "80:80"
    volumes:
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf

🌐 4. NGINX (REVERSE PROXY)
-------------------------------------
cd C:\Users\mashk\IdeaProjects\liferay-work\infra\nginx

y créer :
📄 default.conf
avec le contenu :


server {
    listen 80;
    server_name localhost;

    location / {
        proxy_pass http://liferay:8080;

        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Server $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        proxy_redirect off;
    }
}
